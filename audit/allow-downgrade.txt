#   <key> <from> -> <to>  # <why this is safe, and when it goes back up>

#   setup:GO_VERSION 1.26.11 -> 1.26.10  # 1.26.11 breaks arm64 linking (link to issue); revert when 1.26.12 is out
# '*' may stand for <from> or <to>. Remove the line once the version is back up.
